\documentclass[UTF8,twoside,a4paper,fontset=none]{ctexart}
% overleaf: 锁定 TeX Live 自带字体并跳过字体探测，适合 Overleaf 免费版 10 s 限制。
% 本地 Windows 用户可改为 fontset=windows；通用自动探测可改为 fontset=auto。
\usepackage[fontset=overleaf]{csee}
% gbt7714 基于 natbib，preamble 比 biblatex 轻得多；仍用 refs.bib 管理。
% super/numbers 在 gbt7714 v2.0+ 中已废弃，改用 natbib 的 \setcitestyle 设置上标方括号数字引用。
% sort&compress 由 natbib 提供：合并连续引用，如 \cite{ref1,ref2,ref3} → [1-3]。
\usepackage[sort&compress]{gbt7714}
\setcitestyle{super,open={[},close={]}}
% 参考文献用小五号（9pt）+ 西文 Times 兼容字体，与 biblatex 版和金样一致；
% natbib 默认 thebibliography 沿用 normalsize（10.5pt），会使长刊名多占行。
% 标题、条目间距、悬挂缩进对齐金样（见 csee.sty 的 cseebibliography 与
% biblatex 分支的 \defbibheading / \bibfont / \bibitemsep 设置）。
\renewcommand{\bibfont}{\xiaowu\songti\cseetnr}
\renewcommand{\bibsection}{\par\vspace{4pt}{\noindent\wuhao\heiti 参考文献\par}\vspace{-3.235pt}}
\renewcommand{\bibsep}{0pt}
\renewcommand{\bibhang}{0.63cm}
% natbib 的 numeric 引用模式使用 \NAT@bibsetnum 设置参考文献列表参数，
% 该命令未设置 \topsep，残留 \@listi 的约 8-10pt 顶部间距，导致"参考文献"
% 标题到首条文献间距过宽。此处重定义并清零 \topsep。
\makeatletter
\renewcommand{\NAT@bibsetnum}[1]{\settowidth\labelwidth{\@biblabel{#1}}\setlength{\leftmargin}{\labelwidth}\addtolength{\leftmargin}{\labelsep}\setlength{\itemsep}{\bibsep}\setlength{\parsep}{\z@}\setlength{\topsep}{\z@}}
\makeatother
